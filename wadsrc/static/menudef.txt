LISTMENU "MainMenu"
{
	IfGame(Doom, Chex)
	{
		LineSpacing 16	// This must account for some single-graphic replacements, so it cannot be widened
		StaticPatch 94, 2, "M_DOOM"
		Position 97, 72
		IfOption(ReadThis)
		{
			Position 97, 64
		}
	}
	IfGame(Strife)
	{
		StaticPatch 84, 2, "M_STRIFE"
		Position 97, 45
	}
	IfGame(Heretic)
	{
		StaticPatch 88, 0, "M_HTIC"
		StaticPatch 40, 10, "M_SKL01"
		StaticPatch 232, 10, "M_SKL00"
		Position 110, 56
	}
	IfGame(Hexen)
	{
		StaticPatch 88, 0, "M_HTIC"
		StaticPatch 37, 80, "FBULB0"
		StaticPatch 278, 80, "FBULA0"
		Position 110, 56
	}

	IfGame(Doom, Strife, Chex)
	{
		PatchItem "M_NGAME", "n", "PlayerclassMenu"
		TextItem "$MNU_MULTIPLAYER", "m", "MultiplayerMenu"
		ifOption(SwapMenu)
		{
			PatchItem "M_LOADG", "l", "LoadGameMenu", 0
			PatchItem "M_SAVEG", "s", "SaveGameMenu",0
			PatchItem "M_OPTION","o", "OptionsMenu", 0
		}
		else
		{
			PatchItem "M_OPTION","o", "OptionsMenu", 0
			PatchItem "M_LOADG", "l", "LoadGameMenu", 0
			PatchItem "M_SAVEG", "s", "SaveGameMenu", 0
		}
		ifOption(ReadThis)
		{
			PatchItem "M_RDTHIS","r", "ReadThisMenu", 0
		}
		PatchItem "M_QUITG", "q", "QuitMenu", 0
	}

	IfGame(Heretic, Hexen)
	{
		TextItem "$MNU_NEWGAME", "n", "PlayerclassMenu"
		TextItem "$MNU_MULTIPLAYER", "m", "MultiplayerMenu"
		TextItem "$MNU_OPTIONS", "o", "OptionsMenu"
		TextItem "$MNU_GAMEFILES", "g", "GameFilesMenu"
		TextItem "$MNU_INFO", "i", "ReadThisMenu"
		TextItem "$MNU_QUITGAME", "q", "QuitMenu"
	}
}

		StaticTextCentered 160, 15, "$MNU_CHOOSECLASS"

		StaticText 34, 24, "$MNU_CHOOSECLASS"
		Position 66, 58

// [PERFTRACK] BENCHMARKING -- the owner's own page (2026-09-18: "options -> Benchmarking").
OptionMenu BenchmarkingMenu protected
{
	Title "Benchmarking"

	// [PERFTRACK] TELEMETRY. The performance log above writes a block every N seconds; this writes ONE ROW A FRAME --
	// our CPU split, the GPU groups, the scene's shape, the headset runtime's share, and a reason for every frame over
	// budget. It is off by default and costs one integer test a frame when off. Rows are held in memory and written at
	// the end, so the recorder never measures its own disk writes; a map change flushes, and so does switching it off.
	// Recording a demo at the same time (`record <name>` at the console) makes that demo a benchmark: it carries the
	// build, the whole load order and a state hash per tic, so the same fight can be replayed and compared later.
	// Files land in the save folder unless perf_track_dir says otherwise, and the console names the path.
	StaticText ""
	StaticText "Telemetry", 1
	Option "Record one row a frame", "perf_track", "OnOff"
	Slider "Call a frame slow at (ms, 0 = one display frame)", "perf_track_hitch_ms", 0, 50, 1, 0
	Slider "Stop recording after this many frames", "perf_track_frames", 5000, 100000, 5000, 0
	Option "Put a state hash in recorded demos", "perf_bench_hash", "OnOff"
	SafeCommand "Write what has been recorded so far", "perf_track_flush"
	StaticText ""
	SafeCommand "$OPTMNU_DEFAULTS", "resetcvar perf_track perf_track_hitch_ms perf_track_frames perf_bench_hash"
}

/*=======================================
 *
 * RS FORK -- Voxel Options
 *
 * r_voxels_mode picks which things are voxels: Auto, All, Held & grabbed
 * only, None. Auto (-1, the default) is "held & grabbed only" when a voxel
 * pack came in with a loaded file (r_voxelpack_loaded) and "all" otherwise,
 * so nothing changes without a pack. "Held & grabbed only" means actors with
 * VoxelOverride set, which RS_WorldHands sets on held, laser-locked and
 * gravity-grabbed things, in flight too. r_drawvoxels stays a console switch
 * (archived now); the headset has no console, so the menu row is the mode.
 * This page is also linked from VR Performance.
 *
 * WHY A DISTANCE LIMIT. A decoration voxel pack tags every column, torch,
 * barrel and pickup in the game, so a room that cost a few hundred billboards
 * costs a few hundred model draws instead. Past the limit the ordinary sprite
 * is drawn -- which is exactly what the player saw before the pack was loaded,
 * and is indistinguishable at range.
 *
 * r_voxeldistance_band is hysteresis on that limit: a sprite turns back into
 * a voxel only that much closer, so a thing on the line does not flicker.
 *
 * Held and grabbed things (VoxelOverride) are never distance-culled, and stay
 * voxels in every mode but None. A voxel with no sprite behind it is kept in
 * every mode and at any distance, since refusing it would draw nothing.
 *
 *=======================================*/

// RS fork -- r_voxels_mode. -1 is auto; see the note above and r_utility.cpp.
OptionValue "VoxelModes"
{
	-1, "Auto"
	0, "All"
	1, "Held & grabbed only"
	2, "None"
}
